2. bootstrap/tasks/install.yml

    - name: Ensure symlinks exist for k3s and kubectl in standard PATH locations
      ansible.builtin.file:
        src: "/usr/local/bin/k3s"
        dest: "{{ item }}"
        state: link
        force: true
      loop:
        - "/usr/bin/k3s"
        - "/usr/bin/kubectl"
        - "/usr/local/bin/kubectl"

    - name: Ensure target user kube directory exists
      ansible.builtin.file:
        path: "~{{ ansible_user }}/.kube"
        state: directory
        owner: "{{ ansible_user }}"
        mode: "0700"

    - name: Provision fresh target user kubeconfig
      ansible.builtin.copy:
        src: "/etc/rancher/k3s/k3s.yaml"
        dest: "~{{ ansible_user }}/.kube/config"
        remote_src: true
        owner: "{{ ansible_user }}"
        mode: "0600"
