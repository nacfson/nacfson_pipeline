3. Automated Verification & Acceptance Criteria 

# 1. Verify CI Workflow builds and pushes multi-arch images
gh workflow view build-images.yaml

# 2. Verify PostgreSQL boots healthy and init job completes
kubectl wait --for=condition=ready pod/postgres-0 -n identity --timeout=60s
kubectl wait --for=condition=complete job/postgres-init-job -n identity --timeout=90s

# 3. Verify Proxy Pods running with zero ErrImagePull
kubectl get pods -n vault -l app.kubernetes.io/name=vault-db-proxy
kubectl get pods -n vault -l app.kubernetes.io/name=vault-registry-proxy

# 4. Verify 10/10 Platform Layers converge to Ready: True
kubectl get kustomizations -A

# 5. Run Platform Master Acceptance Suite
./scripts/verify-platform.sh
